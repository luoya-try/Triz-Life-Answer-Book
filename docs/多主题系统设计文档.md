## 二、主题切换器实现

### 2.1 主题管理器 (themeManager.js)

```javascript
// utils/themeManager.js

const THEMES = {
  crayon: {
    name: '卡通风',
    className: 'theme-crayon',
    colors: ['#FF6B9D', '#FFD93D', '#4ECDC4']
  },
  tech: {
    name: '科技风',
    className: 'theme-tech',
    colors: ['#667eea', '#764ba2', '#00d4ff']
  },
  cute: {
    name: '少女风',
    className: 'theme-cute',
    colors: ['#FFB5C5', '#FFE4E1', '#DDA0DD']
  }
};

class ThemeManager {
  constructor() {
    this.currentTheme = 'crayon';
    this.storageKey = 'selected_theme';
  }

  // 初始化主题
  init() {
    // 1. 尝试从本地存储读取
    const savedTheme = wx.getStorageSync(this.storageKey);
    if (savedTheme && THEMES[savedTheme]) {
      this.currentTheme = savedTheme;
    }

    // 2. 应用主题
    this.applyTheme(this.currentTheme);

    // 3. 同步到全局
    const app = getApp();
    app.globalData.currentTheme = this.currentTheme;

    return this.currentTheme;
  }

  // 应用主题
  applyTheme(themeName) {
    if (!THEMES[themeName]) {
      console.error(`主题 ${themeName} 不存在`);
      return;
    }

    this.currentTheme = themeName;

    // 保存到本地存储
    try {
      wx.setStorageSync(this.storageKey, themeName);
    } catch (e) {}

    // 更新页面样式类
    const pages = getCurrentPages();
    pages.forEach(page => {
      if (page.setData) {
        page.setData({
          themeClass: `theme-${themeName}`
        });
      }
    });

    // 触发主题切换事件
    wx.event.emit('themeChange', themeName);

    console.log(`[ThemeManager] 已切换到主题: ${THEMES[themeName].name}`);
  }

  // 切换到下一个主题
  nextTheme() {
    const themeKeys = Object.keys(THEMES);
    const currentIndex = themeKeys.indexOf(this.currentTheme);
    const nextIndex = (currentIndex + 1) % themeKeys.length;
    const nextTheme = themeKeys[nextIndex];

    this.applyTheme(nextTheme);
    return nextTheme;
  }

  // 获取主题配置
  getTheme(themeName) {
    return THEMES[themeName] || THEMES.crayon;
  }

  // 获取当前主题
  getCurrentTheme() {
    return THEMES[this.currentTheme];
  }

  // 获取所有主题列表
  getAllThemes() {
    return Object.entries(THEMES).map(([key, value]) => ({
      id: key,
      ...value
    }));
  }
}

// 单例导出
const themeManager = new ThemeManager();
module.exports = themeManager;
```

### 2.2 主题应用入口

```javascript
// app.js

const themeManager = require('./utils/themeManager.js');

App({
  globalData: {
    currentTheme: 'crayon',
    currentGridId: null,
    gridData: null
  },

  onLaunch() {
    // 初始化主题
    const theme = themeManager.init();
    console.log(`小程序启动，当前主题: ${theme}`);
  },

  // 全局主题切换方法
  switchTheme(themeName) {
    themeManager.applyTheme(themeName);
  }
});
```

### 2.3 页面中使用主题

```javascript
// pages/index/index.js

const themeManager = require('../../utils/themeManager.js');

Page({
  data: {
    themeClass: '',
    themeInfo: null
  },

  onLoad() {
    // 获取主题信息
    const themeInfo = themeManager.getCurrentTheme();
    this.setData({
      themeClass: `theme-${themeManager.currentTheme}`,
      themeInfo: themeInfo
    });

    // 监听主题切换
    wx.event.on('themeChange', (themeName) => {
      const info = themeManager.getTheme(themeName);
      this.setData({
        themeClass: `theme-${themeName}`,
        themeInfo: info
      });
    });
  },

  // 切换主题
  switchTheme() {
    const nextTheme = themeManager.nextTheme();
    wx.showToast({
      title: `${themeManager.getTheme(nextTheme).name}主题`,
      icon: 'success'
    });
  }
});
```

## 六、后续扩展说明

### 6.1 添加新主题步骤

1. 在 `themes/` 目录下创建新主题文件夹
2. 创建 `config.js` 配置文件
3. 创建 `style.css` 样式文件
4. 在 `themeManager.js` 的 `THEMES` 对象中注册
5. 在微信开发者工具中测试主题切换

### 6.2 主题预览功能

```javascript
// 主题预览（不保存）
previewTheme(themeName) {
  const theme = THEMES[themeName];
  if (!theme) return;

  // 临时应用主题
  this.setData({
    themeClass: `theme-${themeName}`,
    isPreview: true
  });

  // 3秒后恢复原主题
  setTimeout(() => {
    if (this.data.isPreview) {
      this.setData({
        themeClass: `theme-${this.currentTheme}`,
        isPreview: false
      });
    }
  }, 3000);
}
```

### 6.3 主题推荐算法

根据场景自动推荐主题：

```javascript
recommendTheme(scenario) {
  const recommendations = {
    'personal_growth': 'crayon',      // 个人成长 → 卡通风
    'technical_problem': 'tech',       // 技术问题 → 科技风
    'lifestyle_share': 'cute',         // 生活分享 → 少女风
    'business_analysis': 'tech',       // 商业分析 → 科技风
    'creative_writing': 'crayon'       // 创意写作 → 卡通风
  };

  return recommendations[scenario] || 'crayon';
}
```
